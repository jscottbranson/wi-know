set(WIKNOW_CONFIG "${CMAKE_CURRENT_LIST_DIR}/../../config/settings.json" CACHE FILEPATH "wi-know JSON settings")
set(WIKNOW_BOARD "" CACHE STRING "Board pin profile: custom, xiao, n32r8")
set(WIKNOW_ROLE "" CACHE STRING "Override role: universal, parent, child, follower")
set(WIKNOW_NODE_ID "" CACHE STRING "Override node ID (children/followers 1..16)")
set(GENERATED_DIR "${CMAKE_CURRENT_BINARY_DIR}/generated")
set(GENERATOR "${CMAKE_CURRENT_LIST_DIR}/../../tools/generate_config.py")
set(GENERATE_ARGS --config "${WIKNOW_CONFIG}" --output "${GENERATED_DIR}/settings.hpp")
if(NOT WIKNOW_BOARD STREQUAL "")
    list(APPEND GENERATE_ARGS --board "${WIKNOW_BOARD}")
endif()
if(NOT WIKNOW_ROLE STREQUAL "")
    list(APPEND GENERATE_ARGS --role "${WIKNOW_ROLE}")
endif()
if(NOT WIKNOW_NODE_ID STREQUAL "")
    list(APPEND GENERATE_ARGS --node-id "${WIKNOW_NODE_ID}")
endif()
if(NOT CMAKE_BUILD_EARLY_EXPANSION)
    file(READ "${WIKNOW_CONFIG}" SETTINGS_JSON)
    string(JSON RULE_DIRECTORY GET "${SETTINGS_JSON}" detection rules_directory)
    get_filename_component(RULE_DIRECTORY "${RULE_DIRECTORY}" ABSOLUTE BASE_DIR "${CMAKE_CURRENT_LIST_DIR}/../..")
    file(GLOB RULE_FILES CONFIGURE_DEPENDS "${RULE_DIRECTORY}/*.json")
    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${WIKNOW_CONFIG}" "${GENERATOR}" ${RULE_FILES})
    execute_process(COMMAND "${PYTHON}" "${GENERATOR}" ${GENERATE_ARGS} COMMAND_ERROR_IS_FATAL ANY)
endif()
idf_component_register(SRCS "main.cpp" "radios.cpp" "peripherals.cpp" INCLUDE_DIRS "." "${GENERATED_DIR}"
    REQUIRES wiknow esp_wifi esp_event esp_netif nvs_flash bt esp_timer esp_driver_uart esp_driver_gpio esp_driver_ledc esp_driver_i2c esp_driver_spi fatfs sdmmc esp_partition wear_levelling esp_psram)
target_compile_features(${COMPONENT_LIB} PUBLIC cxx_std_17)
